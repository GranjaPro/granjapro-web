```typescript
export async function createRevenueFromSale(
  farmId: string,
  saleId: string,
  date: string,
  amount: number,
  description: string
): Promise<FinancialTransaction | null> {
  const supabase = createClient();

  // Check if revenue already exists for this sale (prevent duplicates)
  const { data: existing } = await supabase
    .from("financial_transactions")
    .select("id")
    .eq("farm_id", farmId)
    .eq("reference", `sale:${saleId}`)
    .maybeSingle();

  if (existing) return null; // Already synced

  const { data, error } = await supabase
    .from("financial_transactions")
    .insert({
      farm_id: farmId,
      date,
      type: "revenue",
      category: "sales",
      description,
      amount,
      reference: `sale:${saleId}`,
    })
    .select()
    .single();

  if (error) throw error;
  return data;
}
```

2. The function:
   - Takes farmId, saleId, date, amount, description as parameters
   - Checks for existing entry with `reference = "sale:{saleId}"` using `maybeSingle()` (returns null if not found, no error)
   - If already exists, returns null (idempotent -- safe to call multiple times)
   - If not exists, inserts a new revenue transaction with category "sales" and reference linking back to sale
   - Returns the created FinancialTransaction or null

3. Note: category "sales" is not in the `transactionCategories` constants (which only has expense categories). This is intentional -- revenue categories are not filtered/displayed the same way as expense categories. The `getFinancialSummaryInRange` function only aggregates categories for `type === "expense"`, so "sales" as a revenue category is safe.
